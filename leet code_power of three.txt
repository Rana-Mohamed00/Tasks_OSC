#include <iostream>
#include <cmath>
using namespace std;

bool isPowerOfThree(int n) {
    if (n <= 0) return false;
    double result = log10(n) / log10(3);
    return result == (int)result;
}

int main() {
    int n;
        cin >> n;

    if (isPowerOfThree(n)) {
        cout << "YES";
    } else {
        cout <<"NO";
    }
    return 0;
}
