#include <iostream>
#include <cmath>
using namespace std;
int main()
{
    int t,n,num1,num2;
    bool isPerfect[1000];
    cin>>t;
    
    for(int i=0;i<t;i++){
        isPerfect[i]=true;
        cin>>n;
        cin>>num1;
        for(int j=0;j<n-1;j++){
            cin>>num2;
            if(abs(num1-num2)!=5 && abs(num1-num2) !=7){
                if(isPerfect[i]==false){
                    continue;
                }
                isPerfect[i] = false;
            }
            num1=num2;
        }
        
    }
    for(int i=0;i<t;i++){
        if(isPerfect[i]==true){
            cout<<"YES"<<endl;
        }else{
            cout<<"NO"<<endl;
        }
    }
}
